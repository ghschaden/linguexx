%% \verb in an example body, in the ENVIRONMENT syntax.
%%
%% The manual states a matrix (§ Notes and limitations): \verb works under
%% exe/xlist with an unbraced \ex, including inside an "\a." written within
%% the batch, because that body goes straight to TeX; it works in the dot
%% syntax because the collector reads it specially (verb-dot.tex); and it
%% cannot work in the braced \ex[j]{text} form, whose body is a macro
%% argument (verb-braced.tex); langsci's \ea ... \z streams like exe
%% (verb-langsci.tex).  This pins the first cell, so a change that
%% routed the environment syntax through the collector -- which reads only
%% \verb, \Verb and \lstinline specially, not every verbatim command an
%% exe body takes -- would show here.
%%
%% The payloads carry catcode-active characters, each behind a sentinel of
%% its own so that the extracted token is unique.  "&" is deliberately not
%% among them: pdftotext's -bbox output is XML and escapes it to "&amp;",
%% which would put the escaping, not the verbatim, on trial.
\input{_preamble-gb4e}
\begin{document}
\begin{exe}
\ex VBEXE \verb|exe_%$#{}| tail.
\a. VBSUB \verb|sub_%$#{}| tail.
\z.
\ex VBTWO plain body.
 \begin{xlist}
 \ex VBXL \verb|xlist_%$#{}| tail.
 \end{xlist}
\ex VBESC \verb|esc~^\| tail.
%% Monospace probe: in a fixed-advance font six i's are exactly as wide as
%% six W's; in the body font they are nowhere near.  The \textrm pair below
%% is the sensitivity control for that -- same two strings, so a probe that
%% measured nothing could not pass.
\ex VBMONO \verb|MONOiiiiii| \verb|MONOWWWWWW| tail.
\ex VBROMAN \textrm{ROMNiiiiii} \textrm{ROMNWWWWWW} tail.
\end{exe}
\end{document}
