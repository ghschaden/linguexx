%% \exewidth, which the langsci-gb4e manual documents in its section 7.6:
%% "Use \exewidth{(222)} to set the width reserved for identifiers to
%% three-digits."  It sets this package's own \Exlabelwidth, so the sample
%% given is what the label box is made as wide as.
%%
%% What is NOT provided, and what this case therefore also pins by its
%% absence: upstream widens the box by itself once the numbers pass 98 and
%% 998.  Its only interface is the [manualexewidth] option, which neither
%% document mentions, and the manual's own instruction for a document with
%% three-digit numbers is to call \exewidth.  So a four-digit number here
%% sits in whatever box the document asked for, and WSMALL and WHUGE align.
\input{_preamble-langsci}
\begin{document}
\setcounter{ExNo}{5}
\ea WSMALL a two-digit number.
\z
\setcounter{ExNo}{1000}
\ea WHUGE a four-digit number, in the same box.
\z

\exewidth{(2345)}
\ea WWIDE after \string\exewidth\ asked for a wider one.
\z
\end{document}
