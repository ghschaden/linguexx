%% The two langsci-gb4e options that have a DEFAULT state worth pinning:
%% autoexewidth (on) and the item penalty (the kernel's).  Their other state
%% is langsci-options.tex, and the two cases are only meaningful together --
%% either one alone would pass with the option's code deleted.
%%
%% autoexewidth widens the label box once the numbers reach four digits.  It
%% must WIDEN ONLY: this package's default box is already wider than "(235)",
%% so setting the sample outright -- which is what \exewidth does, and what
%% the first version of this did -- moved a three-digit document's text LEFT
%% of a two-digit one's.  Hence three examples and not two: WBIG is the one
%% that catches it.
\input{_preamble-langsci}
\makeatletter
\begin{document}
\setcounter{ExNo}{5}
\ea WSMALL a two-digit number.
\z
\setcounter{ExNo}{100}
\ea WBIG a three-digit number.
\z
\setcounter{ExNo}{1000}
\ea WHUGE a four-digit number.
\z

\ea WPENALTY the item penalty here is \the\@itempenalty\ by default.
\z
\end{document}
