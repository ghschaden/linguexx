%% The two names langsci-gb4e has never made work, refused by name.

%% [langsci] provides what langsci-gb4e provides, documented or not -- that
%% is the criterion, and the .sty records why.  Its limit is a name with no
%% working behaviour to copy: upstream's xlistabr labels every item
%% "(xnumii." and its qlist raises "No counter 'xnum' defined".  Providing
%% either would mean inventing a meaning for a name whose only precedent is
%% a bug, so both are refused.
%%
%% Refused BY NAME rather than left undefined, so that a document being
%% ported stops at the line that needs changing with a message saying what
%% to write instead.  This case pins one of the two; langsci-names.tex
%% asserts that both are still defined to complain, and that every name that
%% IS provided still exists.  See EXPECT_ERROR.
\input{_preamble-langsci}
\begin{document}
RETIREDPROSE ordinary text.
\begin{exe}
\ex RETIREDHOST the host example.
\begin{xlistabr}\ex RETIREDITEM the item it will never label. \end{xlistabr}
\end{exe}
\end{document}
