%% The user-facing numbering parameters, at values that are NOT their
%% defaults, plus the relative-reference commands that no other case calls.
%%
%% [legacy] exercises some of these, but only at linguex's values; nothing
%% checked that setting them yourself works.  A change that broke
%% customisation while leaving both shipped modes intact would have passed
%% the whole suite.
\input{_preamble}
%% sub-labels: parenthesised letters, bracketed romans
\renewcommand\SubExLBr{(}\renewcommand\SubExRBr{)}
\renewcommand\SubSubExLBr{[}\renewcommand\SubSubExRBr{]}
%% reference dashes: a colon between number and letter, a slash before the
%% roman, and "to" as the range dash
\renewcommand\firstrefdash{:}
\renewcommand\secondrefdash{/}
\renewcommand\rangedash{to}
\begin{document}

\b.\sublabel{c:ii} CUSTII second roman.
\z.\z.

CRANGE \refrange{c:a}{c:b} CROMANRANGE \refrange{c:i}{c:ii}
