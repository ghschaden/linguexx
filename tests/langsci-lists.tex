%% The sub-level numbering variants of langsci-gb4e, and \qlist.
%%
%% Each of these sets two things that have to agree -- the printed label and
%% the /ListNumbering class a screen reader announces -- and only one of them
%% shows on the page.  This case asserts the printed half; langsci-ua.tex
%% asserts the other half on the same constructs, because a list labelled
%% "A." whose class says LowerRoman is well-formed PDF, passes veraPDF, and
%% is simply false.
%%
%% Two of these do NOT reproduce langsci-gb4e's behaviour, deliberately, and
%% the reason is recorded beside each in the .sty: upstream's xlistabr labels
%% every item "(xnumii." and upstream's qlist raises "No counter 'xnum'
%% defined" and labels every item ".".  Both are upstream defects, so what is
%% asserted here is what the names promise.
\input{_preamble-langsci}
\begin{document}
\begin{exe}
\ex LLHOST the host example.
 \begin{xlist}  \ex LLDEFAULT.  \end{xlist}
 \begin{xlista} \ex LLALPH.     \end{xlista}
 \begin{xlistabr}\ex LLABR.     \end{xlistabr}
 \begin{xlisti} \ex LLROMAN.    \end{xlisti}
 \begin{xlistn} \ex LLARABIC.   \end{xlistn}
 \begin{xlistA} \ex LLUPALPH.   \end{xlistA}
 \begin{xlistI} \ex LLUPROMAN.  \end{xlistI}
\end{exe}

%% A variant must not leak its numbering into the list after it: the style is
%% set in the environment's own group and nowhere else.  A plain xlist here,
%% after six variants, has to be lettered again.
\begin{exe}
\ex LLAFTER a second host.
 \begin{xlist} \ex LLAGAIN back to the default numbering. \end{xlist}
\end{exe}

\begin{qlist}
\item LLQA the first question.
\item LLQB the second.
\end{qlist}
\end{document}
