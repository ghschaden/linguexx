%% PDF/UA-2 validation case: roughly the construct surface of tagged.tex,
%% but built as a real accessible document so the suite can run veraPDF on
%% the result instead of leaving it to a manual step before release.
%%
%% What this catches that no other case can: a structure element opened at
%% the wrong moment.  Opening the \glt language Span while still in vertical
%% mode -- rather than deferring it to \everypar, so the paragraph's own P
%% and MC are already open -- produces marked content that straddles its
%% parent.  Every geometric and flat-structure assertion in this suite
%% accepts that happily; veraPDF rejects it on all three profiles.
%%
%% Deliberately NO footnote here, although examples in footnotes are a
%% supported (and delicate) construct.  On this TeX Live a footnote in a
%% tagged UA-2 article emits content that is "neither marked as Artifact nor
%% tagged as real content" under pdflatex and xelatex -- reproducible with a
%% plain \documentclass{article} and no linguexx loaded at all, so it is a
%% kernel/latex-lab matter and not ours to assert on.  Footnote examples stay
%% covered structurally by tagged.tex and, end-to-end, by examples/ua-demo.
\input{_preamble-ua}
\section{UASECT accessible constructs}

UAINTRO ordinary running text precedes the first example.

\ex.\label{ua:one} UAMAIN a numbered example.
\a. UAALPHA a sub-example.
\b. *UABETA a judged sub-example.
\a. UAROMAN a sub-sub-example.
\z.\z.

%% gloss: object-language tier, a Leipzig abbreviation with its expansion,
%% and a free translation in a third language -- the \glt Span under test
\GlossTierLang{1}{de}
\GlossTransLang{en}
\exg. UAOBJ Der Hund bellte.\\ the dog bark.\lpzg{3sg.pst}\\
\glt `UATRANS The dog barked.'

%% The same translation-with-a-language, but inside a roman level that a
%% \z. pops while the example goes on.  That pop is the third exit an
%% example has -- \lx@bodyend and the environment ends are the others --
%% and it was the one that did not close the \glt language Span, which
%% then stayed open across the list close: veraPDF failed all three
%% profiles, nothing showed on the page, and every structure assertion in
%% this file still passed.
%%
%% Three things here are load-bearing, and each was got wrong once while
%% this case was being written.  It must stand AFTER \GlossTransLang above
%% (before it, no Span is opened at all and the case goes green with the
%% bug in place).  The gloss must be at the ROMAN level, since a \z. at
%% the letter level ends the example through \lx@bodyend, which closes the
%% Span anyway.  And the \b. after the pop is not decoration: without it
%% the example ends immediately and \lx@bodyend closes the Span again.
\ex. UAZMAIN an example whose sub-sub-level carries a translation.
\a. UAZALPHA a sub-example.
\a. \gll UAZOBJ Der Hund bellte.\\
        UAZGLOSS the dog barked.\\
\glt `UAZTRANS The dog barked.'
\z.
\b. UAZAFTER a sub-example after the pop.

%% A modified label.  Where the font has no bold small caps (Latin
%% Modern under the unicode engines; T1/cmr under pdflatex has them and
%% takes the plain \textsc path), the caps are made from uppercase
%% letters, and marked content of their own carries the /ActualText that
%% keeps the text layer saying "m" -- a second MC inside the
%% abbreviation's Span, which is exactly the kind of nesting veraPDF is
%% here to judge.
UAMOD a modified abbreviation \lpzg{\textbf{m}.pl} in running text.

%% stacked alternatives, text-mode: no Formula element may appear (that is
%% the PDF/UA-2 failure which forced the 0.12 removal of \altg)
\ex. UAALTN I saw \altn{the cat}{a dog} here.

\exg. UAALTG \altg{Socke}{Tonne} bellte.\\
      altg \altg{sock.\lpzg{sg}}{ton.\lpzg{sg}} bark.\lpzg{pst}\\
\glt `UAALTGT The sock or the ton barked.'

Reference to \ref{ua:one} resolves.
%% ... and a relative reference beside it.  This one is a link of
%% linguexx's own making rather than one hyperref built for a \label,
%% so its Link element and the OBJR under it are ours to get right --
%% and an annotation that is not in the structure tree is a PDF/UA
%% failure, not a cosmetic one.  \Last resolves to the \exg. above.
UAREL relative reference \Last\ resolves too.

\begin{exe}
\ex[??]{UAEXE an item in a batch.}
\end{exe}

%% A free translation set BESIDE the grid rather than under it.  Here rather
%% than in tests/glt-side.tex because this is the case veraPDF is run on,
%% and only veraPDF can answer the question this construct raises: the grid
%% and the translation are two boxes, each carrying marked content of its
%% own, put down inside a paragraph that has already opened some.  Nesting
%% them there is what every veraPDF PROFILE calls compliant -- measured
%% during implementation, 17 records logged against 0 for an ordinary gloss,
%% all three verdicts passing in both cases.  What separates them is the
%% second read of veraPDF's output, which a_ua does.
{\GlossTransSide
\exg. UASIDEOBJ Der Hund bellte.\\ the dog bark.\lpzg{pst}\\
\glt `UASIDETRANS The dog barked, with a translation long enough to take a
      line or two in a column of its own.'
}
