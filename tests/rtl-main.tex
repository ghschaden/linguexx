%% Examples in a right-to-left document must be the MIRROR of examples in
%% a left-to-right one: number against the right margin, text the same
%% distance in from it, a sub-example's letter at the text's edge, a
%% judgment mark hanging to the right of what it judges.  The left-to-right
%% example in the english environment is the reference, measured in the
%% same file, so nothing here hard-codes a length.
%%
%% Two defects this case was written for.  Under XeLaTeX every label sat
%% against its text instead of the margin, 14pt and 8pt inward: the label
%% box was flush LEFT, and bidi leaves [l] physically left where LuaTeX
%% mirrors it (\lx@labelside).  And under both engines an \altn stack read
%% "}stack{": the braces are emitted L, stack, R in source order, a
%% right-to-left paragraph lays that order out right to left, and the
%% pictures inside the boxes are not mirrored (\__lxp_unit:n).  Three
%% stacks: in Hebrew text; in a line of Latin text, which polyglossia
%% reorders left to right under LuaLaTeX and which is what defeated
%% drawing the other side's shape instead; and a solo \altg.  A brace is
%% read off its neighbours' POSITIONS, never their names, because which
%% word ends up on which side is the engines' business.
%%
%% Numbers follow the MAIN language's direction (decided 2026-10-10): the
%% english example ENMAIN is laid out exactly as the Hebrew one is -- number
%% at the right margin, letters and marks to the right -- and only its
%% words run left to right, its short last lines standing against the text
%% edge.  Both are measured against the same mirrored distances.
%%
%% ENNOTEXT's example has no text of its own (\ex. \a. ...): LaTeX keeps
%% its number pending and sets it on the first sub-example's line, and the
%% mirroring must not start a paragraph of its own for it, which left the
%% number alone on a line above (\lx@dir@itemafter).
%%
%% Latin sentinels in Hebrew text: pdftotext reports them as words either
%% engine, and their place in a right-to-left line is what is measured.
\input{_preamble-rtlmain}
\begin{document}
%% The reference is set as a LEFT-TO-RIGHT document would set it, which
%% here takes clearing \@RTLmain around it: an english example in this
%% document is otherwise mirrored like every other (ENMAIN below).
\makeatletter
{\@RTLmainfalse
\begin{english}
\ex. LTRMAIN the reference.
\a. LTRSUBA first.
\b. *LTRSUBB second.
\z.
\end{english}}
\makeatother

\ex. RTLMAIN הילד אכל.
\a. RTLSUBA הילד.
\b. *RTLSUBB אכל.
\z.

\begin{english}
\ex. ENMAIN an English example in the Hebrew document, long enough to run onto a second line of its own so that both lines are measured.
\a. ENSUBA first.
\b. *ENSUBB second.
\z.
\end{english}

\begin{english}
\ex. \a. ENNOTEXT first.
\b. ENNOTEXTB second.
\z.
\end{english}

\ex. אני ראיתי \altn{STKA}{STKB} אתמול.

\ex. BEFORESTK \altn{STKC}{STKD} AFTERSTK.

\ex. אני ראיתי \altg{STKE}{STKF} אתמול.
\end{document}
