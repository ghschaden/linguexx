     aaa bbb ccc \\

     tierfour bracedfour tierfourc \\ \endgl

     hhh iii \\

%% An empty cell, written {}: the gloss of a word that has none, like a gap
%% left by movement.  It is a column of its own, not a double space -- until
%% 2026-09-29 it was dropped, and every gloss after it slid one column
%% left, onto the wrong word.  In the first column too, and next to { },
%% which always worked because the space kept it non-empty.
\ex. \gll EMPA EMPB EMPC EMPD \\
     empa {} empc empd \\
\glt `An empty cell.'

\ex. \gll EMQA EMQB EMQC \\
     {} emqb { } \\
\glt `An empty cell first.'
