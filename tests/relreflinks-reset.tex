%% A reset example counter: two examples, one printed number, and no link.
%%
%% Resetting ExNo makes two examples print the same number.  Until
%% 2026-10-03 they also claimed one hyperref anchor, \theHExNo being built
%% from ExNo alone; the anchors now come from a serial nothing resets, so
%% each example has its own (doc/DEFERRED-DECISIONS.md, "Two examples, one
%% hyperref anchor").  What this case still pins is the promise the
%% relative references make: they ask for a NUMBER, and a number two
%% examples print is ambiguous, so it is withheld from linking exactly as
%% a number no example carries is, and reported in its own words.
%%
%% \setcounter rather than [legacy]'s per-chapter reset, which is the same
%% situation in a book class: one mechanism, and this way the case needs
%% no chapters to show it.
\input{_preamble}
\usepackage{hyperref}
\begin{document}
\ex.\label{rs:one} RSONE first example, number 1.

\setcounter{ExNo}{0}

\ex. RSTWO second example, number 1 again.

\ex. RSTHREE third example, number 2 and unique.

%% (1) is claimed by two examples and must not link; (2) by one and must.
RSDUP \LLast\ RSUNIQ \Last
\end{document}
