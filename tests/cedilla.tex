%% commands \b, \c, \d outside an example, and must still work inside one

%% documented.  Loaded with hyperref, because a \section title is
%% \edef-expanded to build the PDF bookmark and that is where an
%% \@ifnextchar-based dispatch on \c silently dropped the accent.

%% and dies with "Use of \lx@dot@a doesn't match its definition".
\subsection{TITLEcaf\a'e}
